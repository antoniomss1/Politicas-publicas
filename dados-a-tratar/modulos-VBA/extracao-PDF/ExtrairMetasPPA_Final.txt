Sub ExtrairMetasPPA_Final()
    Dim wsOrigem As Worksheet, wsDestino As Worksheet
    Dim linhaOrigem As Long, ultimaLinha As Long, linhaDestino As Long
    Dim conteudoA As String, conteudoB As String, textoCompleto As String
    Dim partes() As String
    
    ' 1. Configuração de Planilhas
    On Error Resume Next
    ' IMPORTANTE: O nome da aba deve ser exatamente PPA (sem espaços)
    Set wsOrigem = ThisWorkbook.Worksheets("PPA")
    
    If wsOrigem Is Nothing Then
        MsgBox "Aba 'PPA' não encontrada! Verifique o nome da aba.", vbCritical
        Exit Sub
    End If
    
    Set wsDestino = ThisWorkbook.Worksheets("Metas")
    If wsDestino Is Nothing Then
        Set wsDestino = ThisWorkbook.Worksheets.Add(After:=wsOrigem)
        wsDestino.Name = "Metas"
    End If
    On Error GoTo 0
    
    wsDestino.Cells.Clear
    wsDestino.Range("A1:F1").Value = Array("Programa", "Unid. Medida", "Esperado K", "Esperado U", "Esperado AK", "Esperado BI")
    linhaDestino = 2
    
    ' Encontra a última linha com qualquer conteúdo
    ultimaLinha = wsOrigem.UsedRange.Rows.Count + wsOrigem.UsedRange.Row - 1

    ' 2. Loop de Extração
    For linhaOrigem = 1 To ultimaLinha
        conteudoA = wsOrigem.Cells(linhaOrigem, 1).Value
        conteudoB = wsOrigem.Cells(linhaOrigem, 2).Value
        
        ' --- BUSCA PROGRAMA ---
        If InStr(1, conteudoA, "Programa:", vbTextCompare) > 0 Or _
           InStr(1, conteudoB, "Programa:", vbTextCompare) > 0 Then
            
            If InStr(1, conteudoA, "Programa:", vbTextCompare) > 0 Then
                textoCompleto = conteudoA
            Else
                textoCompleto = conteudoB
            End If
            
            ' Proteção para evitar o Erro 9 no Split
            partes = Split(textoCompleto, "Programa:")
            If UBound(partes) >= 1 Then
                wsDestino.Cells(linhaDestino, 1).Value = Trim(Split(partes(1), vbLf)(0))
            End If
            
        ' --- BUSCA UNID. MEDIDA ---
        ElseIf InStr(1, conteudoA, "Unid. Medida", vbTextCompare) > 0 Or _
               InStr(1, conteudoB, "Unid. Medida", vbTextCompare) > 0 Then
            
            ' Pega o valor da linha de baixo (conforme o layout do seu PDF)
            wsDestino.Cells(linhaDestino, 2).Value = Trim(Split(wsOrigem.Cells(linhaOrigem + 1, 1).Value, vbLf)(0))
            
        ' --- BUSCA ESPERADO (Lógica das Colunas Alternadas) ---
        ElseIf InStr(1, conteudoA, "Esperado", vbTextCompare) > 0 Then
            ' Se achou na Coluna A: pega de E, R, AF, BB
            wsDestino.Cells(linhaDestino, 3).Value = wsOrigem.Cells(linhaOrigem + 1, "E").Value
            wsDestino.Cells(linhaDestino, 4).Value = wsOrigem.Cells(linhaOrigem + 1, "R").Value
            wsDestino.Cells(linhaDestino, 5).Value = wsOrigem.Cells(linhaOrigem + 1, "AF").Value
            wsDestino.Cells(linhaDestino, 6).Value = wsOrigem.Cells(linhaOrigem + 1, "BB").Value
            linhaDestino = linhaDestino + 1
            
        ElseIf InStr(1, conteudoB, "Esperado", vbTextCompare) > 0 Then
            ' Se achou na Coluna B: pega de K, U, AK, BI
            wsDestino.Cells(linhaDestino, 3).Value = wsOrigem.Cells(linhaOrigem + 1, "K").Value
            wsDestino.Cells(linhaDestino, 4).Value = wsOrigem.Cells(linhaOrigem + 1, "U").Value
            wsDestino.Cells(linhaDestino, 5).Value = wsOrigem.Cells(linhaOrigem + 1, "AK").Value
            wsDestino.Cells(linhaDestino, 6).Value = wsOrigem.Cells(linhaOrigem + 1, "BI").Value
            linhaDestino = linhaDestino + 1
        End If
    Next linhaOrigem

    wsDestino.Columns("A:F").AutoFit
    MsgBox "Extração Finalizada!", vbInformation
End Sub
