Sub ExtrairMetasDetalhadas()
    Dim wsOrigem As Worksheet
    Dim wsDestino As Worksheet
    Dim linhaOrigem As Long, ultimaLinha As Long
    Dim linhaDestino As Long
    Dim valorA As String, valorB As String
    Dim textoPrograma As String
    
    ' 1. Configuração das Planilhas
    Set wsOrigem = ThisWorkbook.ActiveSheet ' Ou use Worksheets("NomeDaSuaAba")
    
    On Error Resume Next
    Set wsDestino = ThisWorkbook.Worksheets("Metas")
    If wsDestino Is Nothing Then
        Set wsDestino = ThisWorkbook.Worksheets.Add(After:=wsOrigem)
        wsDestino.Name = "Metas"
    End If
    On Error GoTo 0
    
    wsDestino.Cells.Clear
    ' Cabeçalhos conforme sua solicitação
    wsDestino.Range("A1:F1").Value = Array("Programa", "Unid. Medida", "Esperado K", "Esperado U", "Esperado AK", "Esperado BI")
    linhaDestino = 2
    
    ultimaLinha = wsOrigem.Cells(wsOrigem.Rows.Count, "A").End(xlUp).Row
    If ultimaLinha < wsOrigem.Cells(wsOrigem.Rows.Count, "B").End(xlUp).Row Then
        ultimaLinha = wsOrigem.Cells(wsOrigem.Rows.Count, "B").End(xlUp).Row
    End If

    ' 2. Percorre a planilha de origem
    For linhaOrigem = 1 To ultimaLinha
        valorA = Trim(wsOrigem.Cells(linhaOrigem, 1).Value)
        valorB = Trim(wsOrigem.Cells(linhaOrigem, 2).Value)
        
        ' --- BUSCA PROGRAMA ---
        If valorA = "Programa" Or valorB = "Programa" Then
            ' Pega apenas a primeira linha da célula na Coluna C (Split por quebra de linha)
            textoPrograma = wsOrigem.Cells(linhaOrigem, 3).Value
            If InStr(textoPrograma, vbLf) > 0 Then
                wsDestino.Cells(linhaDestino, 1).Value = Split(textoPrograma, vbLf)(0)
            Else
                wsDestino.Cells(linhaDestino, 1).Value = textoPrograma
            End If
            
        ' --- BUSCA UNID. MEDIDA ---
        ElseIf valorA = "Unid. Medida" Or valorB = "Unid. Medida" Then
            ' Pega o valor da Coluna C na mesma linha
            wsDestino.Cells(linhaDestino, 2).Value = wsOrigem.Cells(linhaOrigem, 3).Value
            
        ' --- BUSCA ESPERADO ---
        ElseIf valorA = "Esperado" Or valorB = "Esperado" Then
            ' Pega os valores da LINHA SEGUINTE nas colunas K(11), U(21), AK(37) e BI(61)
            wsDestino.Cells(linhaDestino, 3).Value = wsOrigem.Cells(linhaOrigem + 1, "K").Value
            wsDestino.Cells(linhaDestino, 4).Value = wsOrigem.Cells(linhaOrigem + 1, "U").Value
            wsDestino.Cells(linhaDestino, 5).Value = wsOrigem.Cells(linhaOrigem + 1, "AK").Value
            wsDestino.Cells(linhaDestino, 6).Value = wsOrigem.Cells(linhaOrigem + 1, "BI").Value
            
            ' Após encontrar o "Esperado" (que parece ser o último do bloco), pula linha no destino
            linhaDestino = linhaDestino + 1
        End If
    Next linhaOrigem

    wsDestino.Columns("A:F").AutoFit
    MsgBox "Extração de Metas concluída!", vbInformation
End Sub
